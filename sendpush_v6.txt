// TeamFlex Push Edge Function v6 — 역할/이름 기반 타겟 필터링
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
import webpush from "npm:web-push@3.6.7";

const VAPID_PUBLIC_KEY  = "__VPUB__";
const VAPID_PRIVATE_KEY = "__VPRIV__";
const VAPID_SUBJECT     = "mailto:qudrnr81ppp@gmail.com";

webpush.setVapidDetails(VAPID_SUBJECT, VAPID_PUBLIC_KEY, VAPID_PRIVATE_KEY);

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: CORS });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
  );

  try {
    const body = await req.json();

    // ── 구독 저장 모드 (action: "subscribe") ──────────────
    if (body.action === "subscribe") {
      const { endpoint, p256dh, auth, driver_name } = body;
      if (!endpoint || !p256dh || !auth) {
        return new Response(JSON.stringify({ error: "endpoint/p256dh/auth 필수" }), {
          status: 400, headers: { "Content-Type": "application/json", ...CORS },
        });
      }

      await supabase.from("push_subscriptions").delete().eq("endpoint", endpoint);

      const { error: insertErr } = await supabase.from("push_subscriptions").insert({
        endpoint,
        p256dh,
        auth,
        driver_name: driver_name || "",
      });

      if (insertErr) {
        console.error("[save-sub] 삽입 실패:", insertErr.message);
        return new Response(JSON.stringify({ error: insertErr.message }), {
          status: 500, headers: { "Content-Type": "application/json", ...CORS },
        });
      }

      console.log(`[save-sub] ✅ 구독 저장: ${driver_name || "(이름 없음)"}`);
      return new Response(JSON.stringify({ ok: true }), {
        headers: { "Content-Type": "application/json", ...CORS },
      });
    }

    // ── 알림 발송 모드 ─────────────────────────────────────
    // 지원 파라미터:
    //   target_driver_name : 특정 기사 이름 (teams_poller.py push_notify_driver)
    //   target_name        : 특정 기사 이름 (sendPushToAll HTML)
    //   driver_name        : 특정 기사 이름 (구버전 호환)
    //   target_roles       : 역할 배열 ['leader','admin'] (push_notify_leaders / mdPushToLeaders)
    //   ann_id             : 공지 ID (공지 전체 발송용)
    //   data               : 클릭 시 이동 URL 등 추가 데이터
    const {
      title,
      body: msgBody,
      ann_id,
      driver_name,
      target_driver_name,
      target_name,
      target_roles,
      data: extraData,
    } = body;

    // 단일 기사 대상 이름 (우선순위: target_driver_name > target_name > driver_name)
    const singleTarget: string | null = target_driver_name || target_name || driver_name || null;

    let targetNames: string[] | null = null; // null = 전체 발송

    if (target_roles && Array.isArray(target_roles) && target_roles.length > 0) {
      // 역할 기반: drivers 테이블 + push_subscriptions.role 양쪽에서 이름 수집
      //  (관리자처럼 drivers 테이블에 행이 없는 계정도 push_subscriptions.role로 커버)
      const names = new Set<string>();
      const { data: roleDrivers, error: rdErr } = await supabase
        .from("drivers")
        .select("name")
        .in("role", target_roles)
        .eq("status", "approved");
      if (rdErr) console.error("[send-push] drivers 역할 조회 실패:", rdErr.message);
      else (roleDrivers || []).forEach((d: any) => { if (d.name) names.add(d.name); });

      try {
        const { data: roleSubs } = await supabase
          .from("push_subscriptions")
          .select("driver_name")
          .in("role", target_roles);
        // [권위소스=drivers.role] push 구독 role은 역할 변경 후 stale 될 수 있음(예: 팀장→조장).
        //   후보의 실제 drivers.role을 재확인해, drivers에 다른 역할로 등록된 사람은 제외.
        //   (drivers에 행이 아예 없는 계정=관리자 등은 그대로 신뢰)
        const cand = Array.from(new Set((roleSubs || []).map((s: any) => s.driver_name).filter(Boolean)));
        if (cand.length > 0) {
          const { data: candDrivers } = await supabase
            .from("drivers")
            .select("name,role")
            .in("name", cand);
          const driverRole: Record<string, string> = {};
          (candDrivers || []).forEach((d: any) => { if (d.name) driverRole[d.name] = d.role; });
          cand.forEach((nm: string) => {
            const dr = driverRole[nm];
            if (dr === undefined || target_roles.includes(dr)) names.add(nm);
            else console.log(`[send-push] 제외(역할불일치): ${nm} drivers.role=${dr}`);
          });
        }
      } catch (e) {
        console.error("[send-push] push_subscriptions 역할 조회 실패:", String(e));
      }

      targetNames = Array.from(names);
      console.log(`[send-push] 역할 필터 ${JSON.stringify(target_roles)} → ${targetNames.length}명: ${targetNames.join(", ")}`);
    } else if (singleTarget) {
      targetNames = [singleTarget];
      console.log(`[send-push] 단일 기사 타겟: ${singleTarget}`);
    } else {
      // ann_id가 있는 경우만 전체 발송 허용 (공지 알림)
      console.log(`[send-push] ${ann_id ? "공지 전체 발송" : "타겟 없음 — 전체 발송"}`);
    }

    // push_subscriptions 쿼리
    let query = supabase.from("push_subscriptions").select("*");
    if (targetNames && targetNames.length > 0) {
      query = query.in("driver_name", targetNames);
    } else if (targetNames && targetNames.length === 0) {
      // 역할은 있는데 해당 구독자가 없음
      return new Response(JSON.stringify({ sent: 0, message: "대상 구독자 없음" }), {
        headers: { "Content-Type": "application/json", ...CORS },
      });
    }

    const { data: subs, error } = await query;
    if (error) throw error;

    console.log(`[send-push] 발송 대상 구독 수: ${subs?.length ?? 0}`);
    if (!subs || subs.length === 0) {
      return new Response(JSON.stringify({ sent: 0, message: "구독자 없음" }), {
        headers: { "Content-Type": "application/json", ...CORS },
      });
    }

    const pushPayload = JSON.stringify({
      title: title || "TeamFlex 공지",
      body:  msgBody || "새 알림이 있습니다.",
      icon:  "/icons/icon-192.png",
      badge: "/icons/icon-192.png",
      tag:   "teamflex-" + (singleTarget ? "md" : ann_id ? "ann" : "all"),
      data:  extraData || { url: "/TeamFlex_기사포털.html", ann_id: ann_id || "" },
    });

    let sent = 0, failed = 0;
    const deadSubs: number[] = [];

    for (const sub of subs) {
      try {
        await webpush.sendNotification(
          { endpoint: sub.endpoint, keys: { p256dh: sub.p256dh, auth: sub.auth } },
          pushPayload
        );
        sent++;
        console.log(`[send-push] ✅ ${sub.driver_name}`);
      } catch (e: any) {
        console.error(`[send-push] ❌ ${sub.driver_name} → ${e.statusCode} ${e.message}`);
        if (e.statusCode === 410 || e.statusCode === 404) deadSubs.push(sub.id);
        failed++;
      }
    }

    if (deadSubs.length > 0) {
      await supabase.from("push_subscriptions").delete().in("id", deadSubs);
    }
    if (ann_id && !singleTarget && !target_roles) {
      await supabase.from("announcements").update({ push_sent: true }).eq("id", ann_id);
    }

    return new Response(
      JSON.stringify({ sent, failed, total: subs.length }),
      { headers: { "Content-Type": "application/json", ...CORS } }
    );

  } catch (err) {
    console.error("[send-push] 오류:", String(err));
    return new Response(JSON.stringify({ error: String(err) }), {
      status: 500, headers: { "Content-Type": "application/json", ...CORS },
    });
  }
});
